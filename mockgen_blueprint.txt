## 1. System Architecture Overview

### 1.1 High-Level Architecture
```

└────────────────────┬────────────────────────────────────────┘

        ▼            ▼            ▼
┌──────────────┐ ┌──────────┐ ┌─────────────┐
```

## 3. Database Schema (Supabase/PostgreSQL)

```sql
-- Users table (handled by Supabase Auth)
-- Extending with custom profile
CREATE TABLE profiles (
  id UUID REFERENCES auth.users PRIMARY KEY,
  email TEXT UNIQUE NOT NULL,
  full_name TEXT,
  avatar_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Workspaces (Projects)
CREATE TABLE workspaces (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  description TEXT,
  settings JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- API Blueprints (Generated API structure)
CREATE TABLE api_blueprints (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  prompt TEXT NOT NULL,
  blueprint JSONB NOT NULL,  -- Full API structure
  version INTEGER DEFAULT 1,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Endpoints
CREATE TABLE endpoints (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  blueprint_id UUID REFERENCES api_blueprints(id) ON DELETE CASCADE,
  method TEXT NOT NULL,  -- GET, POST, PUT, PATCH, DELETE
  path TEXT NOT NULL,
  description TEXT,
  request_schema JSONB,
  response_schema JSONB,
  sample_response JSONB,
  status_code INTEGER DEFAULT 200,
  delay_ms INTEGER DEFAULT 0,
  error_rate DECIMAL(3,2) DEFAULT 0.0,  -- 0.0 to 1.0
  custom_headers JSONB DEFAULT '{}'::jsonb,
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(workspace_id, method, path)
);

-- Request History (for testing panel)
CREATE TABLE request_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  endpoint_id UUID REFERENCES endpoints(id) ON DELETE SET NULL,
  method TEXT NOT NULL,
  url TEXT NOT NULL,
  headers JSONB,
  body JSONB,
  response_body JSONB,
  response_status INTEGER,
  response_time_ms INTEGER,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Exports
CREATE TABLE exports (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  export_type TEXT NOT NULL,  -- 'openapi', 'express', 'fastapi', etc.
  content TEXT NOT NULL,
  file_name TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_workspaces_user_id ON workspaces(user_id);
CREATE INDEX idx_endpoints_workspace_id ON endpoints(workspace_id);
CREATE INDEX idx_endpoints_method_path ON endpoints(method, path);
CREATE INDEX idx_request_history_workspace_id ON request_history(workspace_id);
CREATE INDEX idx_exports_workspace_id ON exports(workspace_id);
```

## 4. Core Features Implementation

### 4.1 AI Generation Engine

**File: `lib/ai/generator.ts`**

```typescript
export interface GenerationRequest {
  prompt: string;
  provider: 'ollama' | 'lmstudio' | 'groq' | 'openai';
  model: string;
}

export interface APIBlueprint {
  name: string;
  description: string;
  baseUrl: string;
  endpoints: Endpoint[];
  models: DataModel[];
}

export interface Endpoint {
  method: HTTPMethod;
  path: string;
  description: string;
  requestSchema?: JSONSchema;
  responseSchema: JSONSchema;
  sampleResponse: any;
  statusCode: number;
  errorResponses?: ErrorResponse[];
  pagination?: PaginationConfig;
  validation?: ValidationRule[];
}

export class AIGenerator {
  async generate(request: GenerationRequest): Promise<APIBlueprint> {
    // 1. Select provider
    const provider = this.getProvider(request.provider);
    
    // 2. Build system prompt
    const systemPrompt = this.buildSystemPrompt();
    
    // 3. Build user prompt with examples
    const userPrompt = this.buildUserPrompt(request.prompt);
    
    // 4. Call LLM
    const response = await provider.complete({
      systemPrompt,
      userPrompt,
      model: request.model
    });
    
    // 5. Parse and validate response
    const blueprint = this.parseResponse(response);
    
    // 6. Generate sample data
    const enrichedBlueprint = await this.enrichWithSampleData(blueprint);
    
    return enrichedBlueprint;
  }
}
```

### 4.2 Dynamic Mock Server

**File: `app/api/mock/[workspaceId]/[...path]/route.ts`**

```typescript
export async function GET(
  request: Request,
  { params }: { params: { workspaceId: string; path: string[] } }
) {
  return handleMockRequest('GET', params.workspaceId, params.path, request);
}

export async function POST(request: Request, { params }: any) {
  return handleMockRequest('POST', params.workspaceId, params.path, request);
}

// ... same for PUT, PATCH, DELETE

async function handleMockRequest(
  method: string,
  workspaceId: string,
  pathSegments: string[],
  request: Request
) {
  // 1. Construct full path
  const path = '/' + pathSegments.join('/');
  
  // 2. Find matching endpoint
  const endpoint = await findEndpoint(workspaceId, method, path);
  
  if (!endpoint) {
    return new Response('Endpoint not found', { status: 404 });
  }
  
  // 3. Validate request (if schema exists)
  if (endpoint.requestSchema) {
    const body = await request.json();
    const validation = validateRequest(body, endpoint.requestSchema);
    if (!validation.valid) {
      return Response.json(validation.errors, { status: 400 });
    }
  }
  
  // 4. Simulate delay
  if (endpoint.delay_ms > 0) {
    await delay(endpoint.delay_ms);
  }
  
  // 5. Simulate errors (based on error_rate)
  if (Math.random() < endpoint.error_rate) {
    return simulateError(endpoint);
  }
  
  // 6. Generate response
  const response = generateResponse(endpoint);
  
  // 7. Return with custom headers
  return Response.json(response, {
    status: endpoint.status_code,
    headers: endpoint.custom_headers
  });
}
```

## 7. Technical Challenges & Solutions

### Challenge 1: AI Response Consistency
**Problem:** LLMs may generate inconsistent API structures
**Solution:**
- Strict JSON schema in system prompt
- Response validation layer
- Retry logic with refined prompts
- Fallback templates

### Challenge 2: Dynamic Route Handling
**Problem:** Matching incoming requests to stored endpoints with path params
**Solution:**
- Use path-to-regexp library
- Store compiled regex patterns
- Implement path parameter extraction
- Cache matching results

### Challenge 3: Real-time Monaco Editor Performance
**Problem:** Large JSON schemas slow down editor
**Solution:**
- Debounce validation
- Lazy load editor
- Use web workers for validation
- Implement virtualization for large responses

### Challenge 4: Scalable Mock Server
**Problem:** Multiple workspaces with hundreds of endpoints
**Solution:**
- Index endpoints by workspace + method + path
- Use Redis for caching (optional)
- Implement rate limiting
- Add pagination for endpoint lists
